// 부동산, 대출, 금리, 융자 관련 키워드 검증 함수 function isLoanRelatedNews(title, body) { const text = (title + ' ' + body).toLowerCase(); const keywords = ['부동산', '대출', '금리', '융자', '주담대', 'dsr', 'ltv', '전세', '청약', '분양', '보증금', '주택', '임대', '아파트', '규제', '매매']; return keywords.some(keyword => text.includes(keyword)); } async function loadNaverNews() { const q = (document.getElementById('nvQuery').value || '부동산 대출 금리').trim(); if(!q){nStatus('검색어를 입력해 주세요.','err');return;} const url = `/api/news?query=${encodeURIComponent(q)}&display=20&start=1&sort=${document.getElementById('nvSort').value}`; const btn = document.getElementById('nvBtn'); btn.disabled = true; nStatus('융자 관련 최신 뉴스를 불러오는 중…',''); try { let res, data; try { res = await fetch(url); } catch(e) { throw new Error('CORS_OR_NETWORK'); } try { data = await res.json(); } catch(e) { throw new Error('응답 형식 오류'); } if(!res.ok || data.error) throw new Error(data.error || 'API 오류'); const items = (data.items || []).map(it => ({ title: newsClean(it.title), body: newsClean(it.description), link: it.originallink || it.link || '', pub: it.pubDate || '' })); const z = n => String(n).padStart(2,'0'); let added = 0; items.forEach(it => { // 융자/부동산/대출 관련 뉴스가 아니면 제외 if (!isLoanRelatedNews(it.title, it.body)) return; // 중복 링크 방지 if (newsList.some(n => n.link === it.link)) return; const d = new Date(it.pub); const date = isNaN(d) ? '-' : d.getFullYear() + '.' + z(d.getMonth() + 1) + '.' + z(d.getDate()); newsList.push({ id: nextNewsId++, category: newsCategory(it.title + ' ' + it.body), date, title: it.title, body: it.body.length > 180 ? it.body.slice(0, 180) + '…' : it.body, link: it.link, source: 'naver' }); added++; }); newsList.sort((x, y) => String(y.date).localeCompare(String(x.date))); renderNewsList(); nStatus(`✔ 부동산·대출·금리 관련 최신 뉴스 ${added}건을 연동했습니다.`,'ok'); } catch(e) { nStatus('불러오기 실패: ' + e.message, 'err'); } finally { btn.disabled = false; } }
